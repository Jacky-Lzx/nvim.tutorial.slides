\section{Python插件配置}

\begin{frame}[fragile]{nvim-treesitter}
  \begin{columns}
    \begin{column}[t]{0.5\linewidth}
      \begin{itemize}
        \item 基础设置
          \begin{lstlisting}[basicstyle=\tiny\ttfamily]
-- ./lua/plugins/treesitter.lua
{
  "nvim-treesitter/nvim-treesitter",
  opts = {
    auto_install = true,
    ensure_installed = { "c", "vim", "vimdoc", "query", "elixir", "heex", "javascript", "html" },
    sync_install = false,
    highlight = { enable = true },
    indent = { enable = true },
  },
  opts_extend = { "ensure_installed" },
},
          \end{lstlisting}
      \end{itemize}
    \end{column}

    \begin{column}[t]{0.5\linewidth}
      \begin{itemize}
        \item Python语言设置
          \begin{lstlisting}[basicstyle=\tiny\ttfamily]
-- ./lua/plugins/lang/python.lua
{
  "nvim-treesitter/nvim-treesitter",
  optional = true,
  opts = {
    ensure_installed = { "python" },
  },
  opts_extend = { "ensure_installed" },
},
          \end{lstlisting}
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[fragile]{conform.nvim}
  \begin{columns}
    \begin{column}[t]{0.5\linewidth}
      \begin{itemize}
        \item 基础设置
          \begin{lstlisting}[basicstyle=\tiny\ttfamily]
-- ./lua/plugins/lsp.lua
{
  "stevearc/conform.nvim",
  event = "BufWritePre",
  opts = {
    formatters_by_ft = {
      ["_"] = { "trim_whitespace" },
    },
    -- ...
  },
  init = function()
    -- ...
  end,
},
          \end{lstlisting}
      \end{itemize}
    \end{column}

    \begin{column}[t]{0.5\linewidth}
      \begin{itemize}
        \item Python语言设置
          \begin{lstlisting}[basicstyle=\tiny\ttfamily]
-- ./lua/plugins/lang/python.lua
{
  "stevearc/conform.nvim",
  optional = true,
  opts = {
    formatters_by_ft = {
      python = { "ruff_organize_imports", "ruff_format" },
    },
  },
},
          \end{lstlisting}
          \begin{itemize}
              \footnotesize
            \item ruff\_organize\_import: import排序
            \item ruff\_format: 代码格式化
          \end{itemize}
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[fragile]{\link{venv-selector.nvim}{https://github.com/linux-cultist/venv-selector.nvim}}
  \begin{lstlisting}[basicstyle=\tiny\ttfamily]
-- ./lua/plugins/lang/python.lua
{
  "linux-cultist/venv-selector.nvim",
  branch = "regexp", -- Use this branch for the new version
  cmd = "VenvSelect",
  opts = {
    settings = {
      options = {
        notify_user_on_venv_activation = true,
      },
    },
  },
  --  Call config for python files and load the cached venv automatically
  ft = "python",
  keys = { { "<leader>cv", "<CMD>VenvSelect<CR>", desc = "Select VirtualEnv", ft = "python" } },
},
  \end{lstlisting}
\end{frame}
